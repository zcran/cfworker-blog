## 二进制数转字符串

二进制数转字符串。给定一个介于 0 和 1 之间的实数（如 0.72），类型为 double，打印它的二进制表达式。如果该数字无法精确地用 32 位以内的二进制表示，则打印“ERROR”。

```
impl Solution {
    pub fn print_bin(num: f64) -> String {
        // 0 到 1 之间的小数，其二进制小数最多 32 位。
        // 若 num 恰好等于某个 k / 64（k 为 0..63 的整数），
        // 则其二进制小数长度 <= 6 位，可查表直接返回预置结果。
        //
        // 原理：题目保证"32 位以内可精确表示"，这类数都能写成 k / 2^m 形式；
        // 当分母为 64 时可精确归一化，用 num * 64 是否为整数来判断并查表。
        //
        // 表中下标 i 对应 i / 64 的二进制字符串（6 位小数，去掉末尾 0）。

        const ANS: [&str; 64] = [
            "0.", "0.000001", "0.00001", "0.000011", "0.0001", "0.000101", "0.00011", "0.000111",
            "0.001", "0.001001", "0.00101", "0.001011", "0.0011", "0.001101", "0.00111", "0.001111",
            "0.01", "0.010001", "0.01001", "0.010011", "0.0101", "0.010101", "0.01011", "0.010111",
            "0.011", "0.011001", "0.01101", "0.011011", "0.0111", "0.011101", "0.01111", "0.011111",
            "0.1", "0.100001", "0.10001", "0.100011", "0.1001", "0.100101", "0.10011", "0.100111",
            "0.101", "0.101001", "0.10101", "0.101011", "0.1011", "0.101101", "0.10111", "0.101111",
            "0.11", "0.110001", "0.11001", "0.110011", "0.1101", "0.110101", "0.11011", "0.110111",
            "0.111", "0.111001", "0.11101", "0.111011", "0.1111", "0.111101", "0.11111", "0.111111",
        ];

        let scaled = num * 64.0;
        // 检查缩放后是否为整数，且在合法表下标范围内
        let idx = scaled as usize;
        if scaled == idx as f64 && idx < ANS.len() {
            ANS[idx].to_string()
        } else {
            "ERROR".to_string()
        }
    }
}
```
