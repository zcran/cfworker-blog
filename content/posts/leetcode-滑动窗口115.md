## M 个非重叠子数组最大和 I

给你一个长度为 n 的整数数组 nums，以及三个整数 m、l 和 r。

你的任务是从 nums 中选择 至少 一个且 至多 m 个 互不重叠的子数组，并满足：

每个被选择的 子数组 的长度都在 [l, r] 范围内（包含两端）。

所有被选择 子数组 的总和 最大 。

返回你能够取得的 最大 总和。

子数组 是数组中一个连续的 非空 元素序列。


C++
```
class Solution {
public:
    // ================================================================
    // 无界DP函数：当子数组数量不受限制时（m足够大），直接求解最大和
    // ================================================================
    // 参数：
    //   preSum - 前缀和数组，preSum[i] 表示前 i 个元素的和
    //   n      - 数组长度
    //   l      - 子数组的最小长度
    //   r      - 子数组的最大长度
    // 返回：
    //   能获得的最大总和（可以选择任意多个互不重叠的子数组）
    // ================================================================
    long long unboundDP(vector<long long>& preSum, int n, int l, int r) {
        // dp[i]：考虑前 i 个元素时能获得的最大和（不限制子数组数量）
        vector<long long> dp(n + 1, 0);

        // 单调队列，存储索引，维护 dp[j] - preSum[j] 的最大值（单调递减）
        deque<int> dq;

        // 初始化答案为极小值，因为所有数都可能为负数
        long long ans = LLONG_MIN / 2;

        // 遍历每个可能的前缀结束位置 i
        // 从 l 开始，因为子数组至少需要 l 个元素
        for (int i = l; i <= n; i++) {
            // t = i - l 是当前子数组长度恰好为 l 时的起始前一个位置
            // 即子数组范围为 [t+1, i]，长度为 i - t = l
            int t = i - l;

            // ------------------------------------------------------------
            // 步骤1：将新候选起点 t 加入单调队列
            // ------------------------------------------------------------
            // 计算候选值：dp[t] - preSum[t]
            // 物理意义：在位置 t 结束时的"净收益"（调整后价值）
            long long curVal = dp[t] - preSum[t];

            // 维护单调递减队列：如果队尾的值 <= 当前值，弹出队尾
            // 保证队列中 dp[j] - preSum[j] 是单调递减的
            while (!dq.empty() && dp[dq.back()] - preSum[dq.back()] <= curVal) {
                dq.pop_back();
            }
            dq.push_back(t); // 将当前索引加入队尾

            // ------------------------------------------------------------
            // 步骤2：移除超出窗口右边界的索引（子数组长度 > r）
            // ------------------------------------------------------------
            // 窗口范围为 [i-r, i-l]，只保留起点 j 满足 i-r <= j <= i-l
            while (!dq.empty() && dq.front() < i - r) {
                dq.pop_front();
            }

            // ------------------------------------------------------------
            // 步骤3：计算 dp[i]（考虑前 i 个元素的最优值）
            // ------------------------------------------------------------
            // 选项1：不选以 i 结尾的子数组，继承前 i-1 个元素的结果
            dp[i] = dp[i - 1];

            // 选项2：选一个以 i 结尾的子数组，起点为队首索引
            if (!dq.empty()) {
                // 计算总收益 = preSum[i] + max(dp[j] - preSum[j])
                // 其中 j 在窗口 [i-r, i-l] 内
                long long cand = preSum[i] + dp[dq.front()] - preSum[dq.front()];

                // 取两种方案的较大值
                dp[i] = max(dp[i], cand);

                // 更新全局最优解
                ans = max(ans, cand);
            }
        }

        // 返回无限制情况下的最大和
        return ans;
    }

    // ================================================================
    // 主函数：最多选择 m 个互不重叠、长度在 [l, r] 内的子数组，最大化总和
    // ================================================================
    long long maximumSum(vector<int>& nums, int m, int l, int r) {
        int n = nums.size(); // 获取数组长度

        // ------------------------------------------------------------
        // 构建前缀和数组：preSum[i] 表示前 i 个元素的和
        // ------------------------------------------------------------
        vector<long long> preSum(n + 1, 0);
        for (int i = 0; i < n; i++) {
            preSum[i + 1] = preSum[i] + nums[i];
        }

        const long long NEG = LLONG_MIN / 2; // 定义负无穷，用于初始化

        // ------------------------------------------------------------
        // 优化：如果 m 足够大，直接调用无界 DP
        // ------------------------------------------------------------
        // 最多可以选择 n / l 个子数组（每个至少 l 长度）
        // 此时子数组数量限制不起作用
        if (m >= n / l) {
            return unboundDP(preSum, n, l, r);
        }

        // ------------------------------------------------------------
        // 有限制情况：DP 枚举子数组数量
        // ------------------------------------------------------------
        // subPrev[j]：使用 k-1 个子数组时，考虑前 j 个元素的最大和
        // 初始化为 0，表示使用 0 个子数组时和为 0
        vector<long long> subPrev(n + 1, 0);

        long long ans = NEG;      // 全局最优答案
        long long prevAns = NEG;  // 上一轮迭代的答案，用于提前终止

        // 外层循环：枚举使用的子数组数量 k，从 1 到 m
        for (int k = 1; k <= m; k++) {
            // subCur[i]：使用 k 个子数组，且最后一个子数组以 i 结尾的最大和
            vector<long long> subCur(n + 1, NEG);

            // 单调队列，维护 subPrev[j] - preSum[j] 的最大值
            deque<int> dq;

            // 内层循环：遍历每个可能的前缀结束位置 i
            for (int i = l; i <= n; i++) {
                // t = i - l 是当前子数组最小长度时的起始前一个位置
                // 即子数组范围为 [t+1, i]，长度为 i - t
                int t = i - l;

                // --------------------------------------------------------
                // 步骤1：将新候选起点 t 加入单调队列
                // --------------------------------------------------------
                // 计算候选值：subPrev[t] - preSum[t]
                // 这是将新子数组起点放在 t+1 时的"净收益"
                long long val = subPrev[t] - preSum[t];

                // 维护单调递减队列
                while (!dq.empty() && subPrev[dq.back()] - preSum[dq.back()] <= val) {
                    dq.pop_back();
                }
                dq.push_back(t); // 将当前索引加入队列

                // --------------------------------------------------------
                // 步骤2：移除超出窗口右边界的索引（子数组长度 > r）
                // --------------------------------------------------------
                // 窗口范围为 [i-r, i-l]
                while (!dq.empty() && dq.front() < i - r) {
                    dq.pop_front();
                }

                // --------------------------------------------------------
                // 步骤3：如果队列非空，可以形成有效子数组
                // --------------------------------------------------------
                if (!dq.empty()) {
                    // 计算总收益 = preSum[i] + max(subPrev[j] - preSum[j])
                    subCur[i] = preSum[i] + subPrev[dq.front()] - preSum[dq.front()];

                    // 更新全局最优解
                    ans = max(ans, subCur[i]);
                }
            }

            // ------------------------------------------------------------
            // 提前终止优化：如果这一轮的答案没有提升
            // 说明继续增加子数组数量也不会得到更好的结果
            // ------------------------------------------------------------
            if (ans == prevAns) {
                break;
            }
            prevAns = ans; // 记录当前答案用于下一轮比较

            // ------------------------------------------------------------
            // 计算前缀最大值，作为下一轮的 subPrev
            // ------------------------------------------------------------
            // gCur[i]：使用 k 个子数组时，考虑前 i 个元素的最大和（不一定以 i 结尾）
            // 这是对 subCur 取前缀最大值，用于下一轮 DP
            vector<long long> gCur(n + 1, NEG);
            for (int i = 1; i <= n; i++) {
                gCur[i] = max(gCur[i - 1], subCur[i]);
            }

            // 将 gCur 作为下一轮的 subPrev（移动语义，高效转移）
            subPrev = move(gCur);
        }

        // 返回最终的最大和
        return ans;
    }
};
```
